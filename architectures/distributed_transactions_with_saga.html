Distributed Transactions with Saga Pattern

A guide to the Saga pattern, a robust way to manage distributed transactions in microservices, along with runnable Go code examples.

What is a Distributed Transaction?

In a monolithic application, a transaction that involves multiple operations can be handled with a single database transaction. However, in a microservices architecture, operations are distributed across multiple services, each with its own database. A distributed transaction is a set of operations that are guaranteed to be atomic across these multiple services. The Saga pattern is one of the most popular ways to achieve this.

A Saga is a sequence of local transactions where each transaction updates the database of a single service. If any local transaction fails, the Saga executes a series of compensating transactions to undo the work done by the preceding successful transactions.

Two Types of Saga Patterns

The Saga pattern is primarily implemented in two ways: Choreography and Orchestration.

1. Choreography

In a choreography-based Saga, each service produces and listens to events. There is no central orchestrator. Each service decides its own logic and when to trigger the next step of the Saga. It's a decentralized approach, which can be simpler for small Sagas but difficult to manage for complex ones.

2. Orchestration

In an orchestration-based Saga, a central service (the orchestrator) manages the entire transaction. The orchestrator tells each service which local transaction to execute and what to do in case of failure. This approach is more complex to set up but provides better control and simplifies managing the flow of the Saga.

Go Code Examples

Below is a single Go program that simulates both Choreography and Orchestration Sagas for an e-commerce order process. To run this code, simply copy it into a .go file and execute it. 

// This Go program demonstrates two types of Saga patterns: Choreography and Orchestration.
// It simulates an e-commerce transaction involving three services: Order, Payment, and Inventory.
// Each service has a "compensating transaction" to undo its work in case of failure.

package main

import (
	"fmt"
	"math/rand"
	"sync"
	"time"
)

//---------------------------------------------------------------------------------------
// Service definitions
//---------------------------------------------------------------------------------------

// OrderService handles order creation and its compensation (cancelling the order).
type OrderService struct{}

func (OrderService) CreateOrder() error {
	fmt.Println("Order Service: Creating order...")
	// Simulate a successful operation.
	time.Sleep(100 * time.Millisecond)
	fmt.Println("Order Service: Order created.")
	return nil
}

func (OrderService) CancelOrder() {
	fmt.Println("Order Service: Compensating transaction - Cancelling order...")
	time.Sleep(100 * time.Millisecond)
	fmt.Println("Order Service: Order cancelled.")
}

// PaymentService handles payment processing and its compensation (refunding).
type PaymentService struct{}

func (PaymentService) ProcessPayment() error {
	fmt.Println("Payment Service: Processing payment...")
	// Simulate a random failure. 20% chance of failure.
	if rand.Intn(10) > 7 {
		fmt.Println("Payment Service: Payment failed!")
		return fmt.Errorf("payment failed")
	}
	time.Sleep(100 * time.Millisecond)
	fmt.Println("Payment Service: Payment processed successfully.")
	return nil
}

func (PaymentService) RefundPayment() {
	fmt.Println("Payment Service: Compensating transaction - Refunding payment...")
	time.Sleep(100 * time.Millisecond)
	fmt.Println("Payment Service: Payment refunded.")
}

// InventoryService handles inventory updates and its compensation (reverting stock).
type InventoryService struct{}

func (InventoryService) UpdateInventory() error {
	fmt.Println("Inventory Service: Updating inventory...")
	time.Sleep(100 * time.Millisecond)
	fmt.Println("Inventory Service: Inventory updated.")
	return nil
}

func (InventoryService) RevertInventory() {
	fmt.Println("Inventory Service: Compensating transaction - Reverting inventory...")
	time.Sleep(100 * time.Millisecond)
	fmt.Println("Inventory Service: Inventory reverted.")
}

//---------------------------------------------------------------------------------------
// Saga Choreography Pattern
//---------------------------------------------------------------------------------------

func runChoreographySaga() {
	fmt.Println("--- Starting Choreography Saga ---")
	var wg sync.WaitGroup
	wg.Add(3)

	// Channels act as our event bus.
	orderCreatedChan := make(chan bool, 1)
	paymentProcessedChan := make(chan bool, 1)
	inventoryUpdatedChan := make(chan bool, 1)
	compensationChan := make(chan string, 1)

	orderSvc := OrderService{}
	paymentSvc := PaymentService{}
	inventorySvc := InventoryService{}

	// Start the goroutines for each service.
	go func() {
		defer wg.Done()
		if err := orderSvc.CreateOrder(); err != nil {
			compensationChan <- "order"
			return
		}
		orderCreatedChan <- true
	}()

	go func() {
		defer wg.Done()
		// This service waits for the previous event.
		<-orderCreatedChan
		if err := paymentSvc.ProcessPayment(); err != nil {
			compensationChan <- "payment"
			orderSvc.CancelOrder() // Compensate locally.
			return
		}
		paymentProcessedChan <- true
	}()

	go func() {
		defer wg.Done()
		// This service waits for the previous event.
		<-paymentProcessedChan
		if err := inventorySvc.UpdateInventory(); err != nil {
			compensationChan <- "inventory"
			paymentSvc.RefundPayment() // Compensate locally.
			orderSvc.CancelOrder()     // Compensate locally.
			return
		}
		inventoryUpdatedChan <- true
	}()

	// Wait for the saga to complete or fail.
	wg.Wait()
	select {
	case <-compensationChan:
		fmt.Println("Choreography Saga failed and was rolled back.")
	default:
		fmt.Println("Choreography Saga completed successfully.")
	}
	fmt.Println("------------------------------------")
}

//---------------------------------------------------------------------------------------
// Saga Orchestration Pattern
//---------------------------------------------------------------------------------------

// Orchestrator is the central manager of the Saga.
type Orchestrator struct {
	OrderService    OrderService
	PaymentService  PaymentService
	InventoryService InventoryService
}

func (Orchestrator) RunSaga() {
	fmt.Println("--- Starting Orchestration Saga ---")
	// Execute the steps sequentially.
	if err := orchestrator.OrderService.CreateOrder(); err != nil {
		fmt.Println("Orchestration Saga failed: Cannot create order.")
		return
	}
	if err := orchestrator.PaymentService.ProcessPayment(); err != nil {
		fmt.Println("Orchestration Saga failed: Cannot process payment. Initiating compensation.")
		// The orchestrator explicitly calls the compensating transactions.
		orchestrator.OrderService.CancelOrder()
		return
	}
	if err := orchestrator.InventoryService.UpdateInventory(); err != nil {
		fmt.Println("Orchestration Saga failed: Cannot update inventory. Initiating compensation.")
		orchestrator.PaymentService.RefundPayment()
		orchestrator.OrderService.CancelOrder()
		return
	}
	fmt.Println("Orchestration Saga completed successfully.")
	fmt.Println("------------------------------------")
}

var orchestrator = Orchestrator{
	OrderService:    OrderService{},
	PaymentService:  PaymentService{},
	InventoryService: InventoryService{},
}

func main() {
	rand.Seed(time.Now().UnixNano())

	// Run the Choreography Saga simulation.
	runChoreographySaga()

	fmt.Println("\n--- New Transaction ---\n")

	// Run the Orchestration Saga simulation.
	orchestrator.RunSaga()
}
