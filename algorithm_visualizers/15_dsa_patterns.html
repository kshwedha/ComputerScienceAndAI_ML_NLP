1. Sliding Window

Explanation: Efficiently process continuous subarrays by maintaining a dynamic window.

Use Case: Find maximum/minimum in a range, longest substring without repeating characters, etc.

def sliding_window(arr, k):
    window_sum = sum(arr[:k])
    max_sum = window_sum
    for i in range(len(arr) - k):
        window_sum = window_sum - arr[i] + arr[i + k]
        max_sum = max(max_sum, window_sum)
    return max_sum

2. Two Pointers

Explanation: Use two indices to traverse and find relations such as sums or palindromes.

Use Case: Sorted arrays, merging two lists, checking pairs that meet criteria.

def two_pointers(arr, target):
    left, right = 0, len(arr) - 1
    while left < right:
        current_sum = arr[left] + arr[right]
        if current_sum == target:
            return [left, right]
        elif current_sum < target:
            left += 1
        else:
            right -= 1
    return []

3. Fast & Slow Pointers

Explanation: Two pointers at different speeds are used for cycle detection or finding midpoints.

Use Case: Detecting cycles, finding intersection, identifying palindrome lists.

def has_cycle(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            return True
    return False

4. Merge Intervals

Explanation: Sort intervals and merge overlapping ranges to optimize schedules or combine segments.

Use Case: Time management, reservation systems, merging overlapping intervals.

def merge_intervals(intervals):
    intervals.sort(key=lambda x: x[0])
    merged = []
    for interval in intervals:
        if not merged or merged[-1][1] < interval[0]:
            merged.append(interval)
        else:
            merged[-1][1] = max(merged[-1][1], interval[1])
    return merged

5. Cyclic Sort

Explanation: Sort elements by placing them at correct indices, optimizing space and time when the range is known.

Use Case: Find missing or duplicate numbers efficiently without extra space.

def cyclic_sort(arr):
    i = 0
    while i < len(arr):
        correct = arr[i] - 1
        if arr[i] != arr[correct]:
            arr[i], arr[correct] = arr[correct], arr[i]
        else:
            i += 1
    return arr

6. In-place Reversal

Explanation: Reverse or reorder sequences directly within the structure to save space.

Use Case: Rotating arrays, reversing linked lists, checking for palindromes.

def reverse_list(head):
    prev = None
    current = head
    while current:
        nxt = current.next
        current.next = prev
        prev = current
        current = nxt
    return prev

7. Depth-First Search (DFS)

Explanation: Explore paths deeply before backtracking, solving recursive structures in trees or graphs.

Use Case: Maze navigation, puzzle solving, connected components.

def dfs(node, visited):
    if node in visited:
        return
    visited.add(node)
    for neighbor in node.neighbors:
        dfs(neighbor, visited)

8. Breadth-First Search (BFS)

Explanation: Explore neighbor nodes level-by-level using a queue to find shortest paths or explore connectivity.

Use Case: Shortest path problems, traversal with minimal steps.

from collections import deque

def bfs(start_node):
    queue = deque([start_node])
    visited = set([start_node])
    while queue:
        node = queue.popleft()
        for neighbor in node.neighbors:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

11. Greedy

Explanation: Make the best local choice at each step with the hope of finding the global optimum.

Use Case: Interval scheduling, resource allocation, cost minimization.

def greedy(tasks):
    tasks.sort(key=lambda x: x.end_time)
    count = 0
    last_end = float('-inf')
    for task in tasks:
        if task.start >= last_end:
            count += 1
            last_end = task.end
    return count

12. Binary Search

Explanation: Divide search space in half iteratively to efficiently find a target value.

Use Case: Sorted arrays, optimization problems, finding boundaries.

def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = left + (right - left) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1

14. Topological Sort

Explanation: Order nodes so that every directed edge UV means U comes before V.

Use Case: Course scheduling, task dependencies, cycle detection.

from collections import deque

def topological_sort(graph):
    in_degree = {u: 0 for u in graph}
    for u in graph:
        for v in graph[u]:
            in_degree[v] += 1
    queue = deque([u for u in graph if in_degree[u] == 0])
    order = []
    while queue:
        u = queue.popleft()
        order.append(u)
        for v in graph[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)
    return order

15. Trie (Prefix Tree)

Explanation: Store strings efficiently for quick prefix searches or auto-complete operations.

Use Case: Searching dictionary words, prefix-based lookups, text prediction.

class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_word = False

class Trie:
    def __init__(self):
        self.root = TrieNode()

    def insert(self, word):
        node = self.root
        for char in word:
            node = node.children.setdefault(char, TrieNode())
        node.is_word = True
